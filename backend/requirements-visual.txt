# Visual pipeline for slide-only videos.
#
# INSTALL THIS IN A SEPARATE VENV. PaddlePaddle pins numpy and protobuf
# versions that conflict with chromadb, and on Windows it is the single
# most fragile dependency in the project. Keeping it isolated means a
# failed OCR install cannot take the core API down.
#
#   py -m venv .venv-visual
#   .venv-visual\Scripts\activate
#   pip install -r requirements-visual.txt
